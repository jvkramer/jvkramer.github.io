% ============================================================================
%  Clean Beamer Presentation Template
%  -----------------------------------
%  Inspired by the Banco de Portugal slide deck aesthetic.
%
%  HOW TO CHANGE THE COLOUR SCHEME:
%    Edit the three colours defined under "COLOUR SCHEME" below.
%    - PrimaryColor : vertical bar, titles, section labels
%    - AccentColor  : subtle highlights (progress-bar background)
%    - TextColor    : body text
%
%  HOW TO CHANGE THE LOGO:
%    Replace the file referenced in \logofile (default: logo.png).
%    A small version is placed top-right on every content slide.
%
%  HOW TO CHANGE THE TITLE-SLIDE IMAGE:
%    Pass a different file to \titleimage.
%
%  Compile with:  pdflatex  (or lualatex / xelatex for system fonts)
% ============================================================================

\documentclass[aspectratio=169, 11pt]{beamer}

% === packages ===============================================================
\usepackage[T1]{fontenc}
\usepackage{lmodern}              % clean serif fallback
\usepackage{tikz}
\usepackage{graphicx}
\usepackage{calc}
\usepackage{etoolbox}
\usetikzlibrary{calc}

% === remove beamer chrome ===================================================
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{footline}{}
\setbeamertemplate{headline}{}

% ============================================================================
%  COLOUR SCHEME  –  edit these three lines to rebrand
% ============================================================================
\definecolor{PrimaryColor}{RGB}{163, 127, 68}   % gold / dark amber
\definecolor{AccentColor}{RGB}{220, 200, 160}    % light gold (bar bg)
\definecolor{TextColor}{RGB}{60, 60, 60}         % dark grey body text
\definecolor{LightGray}{RGB}{245, 245, 245}      % background tint

% === apply colours to beamer elements =======================================
\setbeamercolor{normal text}{fg=TextColor}
\setbeamercolor{frametitle}{fg=PrimaryColor}
\setbeamercolor{itemize item}{fg=PrimaryColor}
\setbeamercolor{itemize subitem}{fg=PrimaryColor}
\setbeamercolor{enumerate item}{fg=PrimaryColor}
\setbeamercolor{enumerate subitem}{fg=PrimaryColor}
\setbeamercolor{block title}{fg=white, bg=PrimaryColor}
\setbeamercolor{block body}{bg=LightGray}

\setbeamerfont{frametitle}{size=\Large, series=\bfseries}
\setbeamerfont{framesubtitle}{size=\normalsize, series=\mdseries}

% === bullet style ===========================================================
\setbeamertemplate{itemize item}{\small\raise1pt\hbox{\textbullet}}
\setbeamertemplate{itemize subitem}{\scriptsize\raise1pt\hbox{--}}

% ============================================================================
%  USER METADATA  –  fill these in for your talk

\newcommand{\presentationtitle}{Presentation Title}
\newcommand{\presentationsubtitle}{Subtitle Goes Here}
\newcommand{\presentationauthor}{Author Name}
\newcommand{\presentationrole}{Role / Affiliation}
\newcommand{\presentationdate}{12 Feb.\ 2026}
\newcommand{\presentationlocation}{City}
\newcommand{\logofile}{logo.png}       % path to your institution's logo
\newcommand{\titleimage}{titlephoto.jpg} % path to title-slide photo

% store total frame count for progress bar
\newcounter{totalframes}

% ============================================================================
%  PROGRESS BAR (free-floating thin vertical bar, left side)
% ============================================================================
%  A narrow bar floats on the left margin.  The filled portion grows
%  from BOTTOM to TOP as you advance through the presentation.
% ============================================================================
\newcommand{\barwidth}{2pt}             % thin, like the original
\newcommand{\barxoffset}{0.38cm}        % distance from left edge
\newcommand{\bartopmargin}{0.7cm}       % gap from top of slide
\newcommand{\barbotmargin}{0.7cm}       % gap from bottom of slide

\newcommand{\drawprogressbar}{%
  \begin{tikzpicture}[remember picture, overlay]
    % coordinates for the bar endpoints
    \coordinate (barbot) at ([xshift=\barxoffset, yshift=\barbotmargin]current page.south west);
    \coordinate (bartop) at ([xshift=\barxoffset, yshift=-\bartopmargin]current page.north west);
    % background bar (full length, light colour)
    \fill[AccentColor, rounded corners=0.5pt]
      ([xshift=-\barwidth/2]barbot) rectangle ([xshift=\barwidth/2]bartop);
    % filled portion (progress, strong colour)
    \ifnum\value{totalframes}>0
      \pgfmathsetmacro{\progress}{%
        (\insertframenumber-1) / (\value{totalframes}-1)
      }
    \else
      \pgfmathsetmacro{\progress}{0}
    \fi
    \path let \p1 = (barbot), \p2 = (bartop),
              \n1 = {\y2-\y1} in
      coordinate (barfill) at ([yshift={\progress*\n1}]barbot);
    \fill[PrimaryColor, rounded corners=0.5pt]
      ([xshift=-\barwidth/2]barbot) rectangle ([xshift=\barwidth/2]barfill);
  \end{tikzpicture}%
}

% ============================================================================
%  LOGO HELPER  –  small logo, top-right corner
% ============================================================================
\newcommand{\drawlogo}{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north east, inner sep=0pt]
      at ([xshift=-0.35cm, yshift=-0.35cm]current page.north east)
      {\IfFileExists{\logofile}%
        {\includegraphics[height=0.55cm]{\logofile}}%
        {%
          % Placeholder when logo file is missing
          \tikz\node[rounded corners=2pt, fill=PrimaryColor,
                      text=white, font=\tiny\bfseries,
                      minimum height=0.55cm, minimum width=1.4cm]{LOGO};%
        }%
      };
  \end{tikzpicture}%
}

% ============================================================================
%  FOOTER  –  presentation title (left) · date + slide number (right)
% ============================================================================
\newcommand{\drawfooter}{%
  \begin{tikzpicture}[remember picture, overlay]
    % left: presentation title
    \node[anchor=south west, inner sep=0pt,
          font=\tiny\color{TextColor!60}]
      at ([xshift=0.5cm, yshift=0.25cm]current page.south west)
      {\MakeUppercase{\presentationtitle}};
    % right: date | page
    \node[anchor=south east, inner sep=0pt,
          font=\tiny\color{TextColor!60}]
      at ([xshift=-0.5cm, yshift=0.25cm]current page.south east)
      {\presentationdate\enspace|\enspace\insertframenumber\,/\,\inserttotalframenumber};
  \end{tikzpicture}%
}

% ============================================================================
%  CONTENT-SLIDE BACKGROUND TEMPLATE
% ============================================================================
\setbeamertemplate{background}{%
  % Only draw on non-title frames
  \ifnum\insertframenumber>1\relax
    \drawprogressbar
    \drawlogo
    \drawfooter
    % thin top rule
    \begin{tikzpicture}[remember picture, overlay]
      \draw[PrimaryColor!30, line width=0.4pt]
        ([yshift=-1.15cm, xshift=0.5cm]current page.north west) --
        ([yshift=-1.15cm, xshift=-0.5cm]current page.north east);
    \end{tikzpicture}
  \fi
}

% ============================================================================
%  FRAME TITLE FORMAT  –  small running header + large title
% ============================================================================
\setbeamertemplate{frametitle}{%
  \vskip2pt
  {\tiny\color{TextColor!50}\MakeUppercase{\presentationtitle}}\\[-2pt]
  \rule{0pt}{0.55cm}%  vertical strut
  {\usebeamerfont{frametitle}\usebeamercolor[fg]{frametitle}\insertframetitle}%
  \ifx\insertframesubtitle\empty\else
    \\[2pt]{\usebeamerfont{framesubtitle}\color{TextColor!70}\insertframesubtitle}%
  \fi
}

% ============================================================================
%  TITLE PAGE TEMPLATE
% ============================================================================
\defbeamertemplate*{title page}{custom}[1][]{%
  % We build the entire title page with TikZ for precise placement.
  \begin{tikzpicture}[remember picture, overlay]
    %
    % --- left half: photo -------------------------------------------------
    \IfFileExists{\titleimage}{%
      \node[anchor=west, inner sep=0pt] at (current page.west)
        {\includegraphics[height=\paperheight, width=0.48\paperwidth,
                          keepaspectratio=false]{\titleimage}};
    }{%
      % Placeholder gradient when image file is missing
      \fill[left color=PrimaryColor!25, right color=PrimaryColor!5]
        (current page.south west) rectangle
        ([xshift=0.48\paperwidth]current page.north west);
      \node[font=\small\color{PrimaryColor!60}, rotate=90]
        at ([xshift=0.24\paperwidth]current page.center |- current page.center)
        {Set \texttt{\textbackslash titleimage} to your photo};
    }
    %
    % --- subtle divider line ----------------------------------------------
    \draw[PrimaryColor!30, line width=0.6pt]
      ([xshift=0.48\paperwidth]current page.south west) --
      ([xshift=0.48\paperwidth]current page.north west);
    %
    % --- right half: text -------------------------------------------------
    \node[anchor=north west, text width=0.42\paperwidth,
          inner sep=0pt]
      at ([xshift=0.52\paperwidth, yshift=-1.8cm]current page.north west)
      {%
        {\color{PrimaryColor}\LARGE\bfseries
         \presentationtitle\par}
        \vskip6pt
        {\color{PrimaryColor!80}\large \presentationsubtitle\par}
      };
    %
    % --- date / location --------------------------------------------------
    \node[anchor=south west, inner sep=0pt,
          text width=0.42\paperwidth]
      at ([xshift=0.52\paperwidth, yshift=2.2cm]current page.south west)
      {%
        {\small\color{TextColor!70}%
         \MakeUppercase{\presentationdate}
         \enspace|\enspace
         \MakeUppercase{\presentationlocation}\par}
        \vskip4pt
        {\small\color{TextColor}%
         \MakeUppercase{\presentationauthor}
         \enspace|\enspace
         \MakeUppercase{\presentationrole}\par}
      };
    %
    % --- logo on title page (bottom right) --------------------------------
    \node[anchor=south east, inner sep=0pt]
      at ([xshift=-0.6cm, yshift=0.6cm]current page.south east)
      {\IfFileExists{\logofile}%
        {\includegraphics[height=0.7cm]{\logofile}}%
        {\tikz\node[rounded corners=2pt, fill=PrimaryColor,
                    text=white, font=\tiny\bfseries,
                    minimum height=0.7cm, minimum width=1.6cm]{LOGO};}%
      };
    %
    % --- vertical bar (static, no progress on title slide) ----------------
    \coordinate (tbarbot) at ([xshift=\barxoffset, yshift=\barbotmargin]current page.south west);
    \coordinate (tbartop) at ([xshift=\barxoffset, yshift=-\bartopmargin]current page.north west);
    \fill[AccentColor, rounded corners=0.5pt]
      ([xshift=-\barwidth/2]tbarbot) rectangle ([xshift=\barwidth/2]tbartop);
  \end{tikzpicture}
}

% ============================================================================
%  SECTION PAGES  –  optional, clean divider
% ============================================================================
\AtBeginSection[]{%
  \begin{frame}[plain,noframenumbering]
    \drawprogressbar
    \begin{tikzpicture}[remember picture, overlay]
      \node[anchor=center, text width=0.7\paperwidth, align=center]
        at ([yshift=0.4cm]current page.center)
        {{\color{PrimaryColor}\Huge\bfseries \insertsection}};
      \draw[PrimaryColor, line width=1pt]
        ([xshift=3cm, yshift=-0.4cm]current page.center) --
        ([xshift=-3cm, yshift=-0.4cm]current page.center);
    \end{tikzpicture}
  \end{frame}
}

\section{Introduction}

\begin{frame}{How to Use This Template}
  \textcolor{PrimaryColor}{\textbf{Getting started}} \\[6pt]
  \begin{itemize}
    \item Edit the metadata block near the top of the \texttt{.tex} file
    \item Replace \texttt{logo.png} and \texttt{titlephoto.jpg} with your own files
    \item Change \texttt{PrimaryColor}, \texttt{AccentColor}, and
          \texttt{TextColor} to match your institution
  \end{itemize}

  \vskip12pt
  \textcolor{PrimaryColor}{\textbf{Compilation}} \\[6pt]
  \begin{itemize}
    \item Compile twice so the progress bar picks up the total frame count
    \item Works with \texttt{pdflatex}, \texttt{lualatex}, or \texttt{xelatex}
  \end{itemize}
\end{frame}
